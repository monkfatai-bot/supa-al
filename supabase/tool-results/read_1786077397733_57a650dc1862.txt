     1→/**
     2→ * Supa AI — Chat orchestration service (Phase 3).
     3→ *
     4→ * The single entry point for AI chat. Orchestrates:
     5→ *
     6→ *   - Resolving provider + model (from request → conversation → env default).
     7→ *   - Validating the context window against the model catalog.
     8→ *   - Checking user credits (throws {@link PaymentError} when ≤ 0).
     9→ *   - Rate-limiting with the `AI_GENERATION` preset.
    10→ *   - Recording the user message to `ai_messages`.
    11→ *   - Calling `ai.chatStream()` with **failover**: on `AIProviderError`,
    12→ *     try the next model in {@link ModelManager.getFailoverChain}.
    13→ *   - Yielding chunks to the caller (the API route pipes them as SSE).
    14→ *   - On success: recording the assistant message with token usage + cost +
    15→ *     latency, recording usage to `ai_usage`, deducting credits.
    16→ *   - On error: recording the assistant message with `error_message` +
    17→ *     `finish_reason='error'`, updating `provider_health`, yielding an
    18→ *     error chunk.
    19→ *
    20→ * The streaming methods are TRUE async generators — the API route wraps them
    21→ * in a `ReadableStream` via {@link createSseResponse}.
    22→ *
    23→ * @module @/lib/chat/chat-service
    24→ */

   108→class ChatService {
   109→  constructor(private readonly admin: AdminSupabaseClient) {}
   110→
   111→  /**
   112→   * Stream an AI response for a new user message. The last entry in
   113→   * `messages` is the new user message; it is recorded to the DB before the
   114→   * AI call. Yields {@link ChatStreamChunk}s as they arrive from the
   115→   * provider, with failover across models in the catalog.
   116→   *
   117→   * Steps:
   118→   *   1. Resolve provider + model.
   119→   *   2. Validate context window.
   120→   *   3. Check user credits.
   121→   *   4. Rate-limit (AI_GENERATION preset).
   122→   *   5. Record the user message.
   123→   *   6. Call `ai.chatStream()` with failover.
   124→   *   7. Yield chunks.
   125→   *   8. On success: record assistant message + usage + deduct credits.
   126→   *   9. On error: record assistant error message + update health + yield error chunk.
   127→   */
   128→  async *streamResponse(
   129→    input: StreamResponseInput,
   130→  ): AsyncIterable<ChatStreamChunk> {
   131→    const { conversationId, userId, messages } = input;
   132→    const log = logger.child({ userId, conversationId });
   133→
   134→    // Validate input shape: messages must be non-empty and end with a user turn.
   135→    const lastMessage = messages[messages.length - 1];
   136→    if (!lastMessage || lastMessage.role !== "user") {
   137→      throw new ValidationError(
   138→        "streamResponse requires a non-empty messages array ending with a user message.",
   139→      );
   140→    }
   141→
   142→    // Fetch the conversation (verifies ownership + lets us default provider/model).
   143→    const conversationService = await createConversationService();
   144→    const conversation = await conversationService.get(userId, conversationId);
   145→    if (!conversation) {
   146→      throw new NotFoundError("Conversation", conversationId);
   147→    }
   148→
   149→    const { provider, model, systemPrompt } = this.resolveProviderModel(
   150→      input,
   151→      conversation,
   152→    );
   153→
   154→    // Validate context window.
   155→    const estimatedTokens = this.estimateTokens(messages, systemPrompt);
   156→    const ctxCheck = modelManager.validateContext(provider, model, estimatedTokens);
   157→    if (!ctxCheck.ok) {
   158→      throw new ValidationError(
   159→        `Conversation exceeds the ${model} context window (${ctxCheck.actual} > ${ctxCheck.limit} tokens). Trim the conversation and try again.`,
   160→        { provider, model, limit: ctxCheck.limit, actual: ctxCheck.actual },
   161→      );
   162→    }
   163→
   164→    // Check credits BEFORE the AI call.
   165→    const creditsService = createCreditsService();
   166→    const balance = await creditsService.checkBalance(userId);
   167→    if (!balance.sufficient) {
   168→      throw new PaymentError("Insufficient credits. Top up your balance to continue.");
   169→    }
   170→
   171→    // Rate-limit.
   172→    await rateLimiter.consumePreset(
   173→      `ai:${userId}`,
   174→      RATE_LIMIT_PRESETS.AI_GENERATION,
   175→    );
   176→
   177→    // Record the user message to DB (best-effort — a failure here shouldn't
   178→    // block the AI call, but we DO want it persisted for history).
   179→    const messageService = await createMessageService();
   180→    let userMessageId: string | null = null;
   181→    try {
   182→      const userMessage = await messageService.create(conversationId, userId, {
   183→        role: "user",
   184→        content: lastMessage.content,
   185→        provider: provider,
   186→        model: model,
   187→      });
   188→      userMessageId = userMessage.id;
   189→    } catch (err) {
   190→      log.warn("failed to persist user message", { error: String(err) });
   191→    }
   192→
   193→    // Run the AI stream with failover.
   194→    yield* this.runAiStream({
   195→      conversationId,
   196→      userId,
   197→      messages,
   198→      provider,
   199→      model,
   200→      systemPrompt,
   201→      temperature: input.temperature,
   202→      maxTokens: input.maxTokens,
   203→      parentMessageId: userMessageId,
   204→    });
   205→  }
   206→
   207→  /**
   208→   * Regenerate the assistant response for a message. Re-runs the
   209→   * conversation from the parent of `messageId`, creating a new branch
   210→   * (the new assistant message's `parent_message_id` is set to the
   211→   * original user message's id).
   212→   */
   213→  async *regenerate(input: RegenerateInput): AsyncIterable<ChatStreamChunk> {
   214→    const { userId, messageId } = input;
   215→    const log = logger.child({ userId, messageId });
   216→
   217→    // Fetch the message (verifies ownership).
   218→    const messageService = await createMessageService();
   219→    const message = await messageService.getMessageForUser(userId, messageId);
   220→    if (message.role !== "assistant") {
   221→      throw new ValidationError("Only assistant messages can be regenerated.");
   222→    }
   223→
   224→    const conversationId = message.conversation_id;
   225→    const conversationService = await createConversationService();
   226→    const conversation = await conversationService.get(userId, conversationId);
   227→    if (!conversation) {
   228→      throw new NotFoundError("Conversation", conversationId);
   229→    }
   230→
   231→    // Resolve provider/model: explicit → conversation → env default.
   232→    const providerOverride = input.provider ?? conversation.provider ?? undefined;
   233→    const modelOverride = input.model ?? conversation.model ?? undefined;
   234→    const resolved = this.resolveProviderModelFromStrings(
   235→      providerOverride,
   236→      modelOverride,
   237→    );
   238→
   239→    // Validate context window.
   240→    const parentMessageId = message.parent_message_id;
   241→    if (!parentMessageId) {
   242→      throw new ValidationError(
   243→        "Cannot regenerate a message without a parent (no conversation history to re-run).",
   244→      );
   245→    }
   246→
   247→    // Build the messages array from the conversation history up to and
   248→    // including the parent (the user message that prompted the original
   249→    // assistant response). We do NOT include the message being regenerated.
   250→    const history = await messageService.list(conversationId, userId, {
   251→      limit: MAX_CONTEXT_MESSAGES,
   252→    });
   253→    const parentIdx = history.findIndex((m) => m.id === parentMessageId);
   254→    if (parentIdx < 0) {
   255→      throw new NotFoundError("Parent message", parentMessageId);
   256→    }
   257→    const uptoParent = history.slice(0, parentIdx + 1);
   258→    const messages = this.buildMessagesFromRows(uptoParent, conversation.system_prompt);
   259→
   260→    const estimatedTokens = this.estimateTokens(messages, conversation.system_prompt ?? undefined);
   261→    const ctxCheck = modelManager.validateContext(resolved.provider, resolved.model, estimatedTokens);
   262→    if (!ctxCheck.ok) {
   263→      throw new ValidationError(
   264→        `Conversation exceeds the ${resolved.model} context window (${ctxCheck.actual} > ${ctxCheck.limit} tokens).`,
   265→        { provider: resolved.provider, model: resolved.model },
   266→      );
   267→    }
   268→
   269→    // Credits + rate-limit (same as streamResponse).
   270→    const creditsService = createCreditsService();
   271→    const balance = await creditsService.checkBalance(userId);
   272→    if (!balance.sufficient) {
   273→      throw new PaymentError("Insufficient credits. Top up your balance to continue.");
   274→    }
   275→    await rateLimiter.consumePreset(
   276→      `ai:${userId}`,
   277→      RATE_LIMIT_PRESETS.AI_GENERATION,
   278→    );
   279→
   280→    yield* this.runAiStream({
   281→      conversationId,
   282→      userId,
   283→      messages,
   284→      provider: resolved.provider,
   285→      model: resolved.model,
   286→      systemPrompt: conversation.system_prompt ?? undefined,
   287→      temperature: input.temperature,
   288→      maxTokens: input.maxTokens,
   289→      parentMessageId,
   290→    });
   291→
   292→    void log;
   293→  }

   299→  /**
   300→   * Core streaming loop shared by {@link streamResponse} and {@link regenerate}.
   301→   * Runs the AI stream with failover, accumulates the full response, records
   302→   * the assistant message + usage, deducts credits, and updates provider
   303→   * health. Yields every chunk to the caller.
   304→   *
   305→   * On a fatal error (all models in the failover chain fail), yields an
   306→   * error chunk and persists an assistant message with `finish_reason='error'`.
   307→   */
   308→  private async *runAiStream(params: {
   309→    conversationId: string;
   310→    userId: string;
   311→    messages: ChatMessage[];
   312→    provider: AIProvider;
   313→    model: string;
   314→    systemPrompt?: string;
   315→    temperature?: number;
   316→    maxTokens?: number;
   317→    parentMessageId: string | null;
   318→  }): AsyncIterable<ChatStreamChunk> {
   319→    const {
   320→      conversationId,
   321→      userId,
   322→      messages,
   323→      provider,
   324→      model,
   325→      systemPrompt,
   326→      temperature,
   327→      maxTokens,
   328→      parentMessageId,
   329→    } = params;
   330→
   331→    const log = logger.child({ userId, conversationId, provider, model });
   332→
   333→    // Prepend the system prompt (default if none) to the messages array.
   334→    const finalMessages: ChatMessage[] = [
   335→      { role: "system", content: systemPrompt ?? DEFAULT_SYSTEM_PROMPT },
   336→      ...messages,
   337→    ];
   338→
   339→    const req = {
   340→      messages: finalMessages,
   341→      model,
   342→      temperature,
   343→      max_tokens: maxTokens,
   344→      stream: true as const,
   345→    };
   346→
   347→    // Failover chain: preferred model first, then same-provider alternatives,
   348→    // then cross-provider alternatives.
   349→    const chain = modelManager.getFailoverChain(provider, model);
   350→    const available = ai.listAvailable();
   351→    const candidates = chain.filter((m) => available.includes(m.provider));
   352→
   353→    if (candidates.length === 0) {
   354→      // No configured provider in the chain — record an error message and yield.
   355→      const errMsg = `No AI provider is configured. Set at least one API key (e.g. OPENAI_API_KEY) to use chat.`;
   356→      yield* this.recordAndYieldError(params, errMsg);
   357→      return;
   358→    }
   359→
   360→    let accumulated = "";
   361→    let inputTokens = 0;
   362→    let outputTokens = 0;
   363→    let totalTokens = 0;
   364→    let finishReason: ChatStreamChunk["finish_reason"] | undefined;
   365→    let usedModel: ManagedModel | null = null;
   366→    let streamError: Error | null = null;
   367→    const startedAt = Date.now();
   368→
   369→    // Try each candidate model in turn.
   370→    for (const candidate of candidates) {
   371→      try {
   372→        // Skip unconfigured providers (defensive — `available` already filtered).
   373→        if (!available.includes(candidate.provider)) continue;
   374→
   375→        const stream = ai.chatStream(req, {
   376→          provider: candidate.provider,
   377→          userId,
   378→          feature: "chat",
   379→        });
   380→
   381→        for await (const chunk of stream) {
   382→          if (chunk.delta) accumulated += chunk.delta;
   383→          if (chunk.usage) {
   384→            inputTokens = chunk.usage.prompt_tokens ?? inputTokens;
   385→            outputTokens = chunk.usage.completion_tokens ?? outputTokens;
   386→            totalTokens = chunk.usage.total_tokens ?? (inputTokens + outputTokens);
   387→          }
   388→          if (chunk.finish_reason) finishReason = chunk.finish_reason;
   389→          yield chunk;
   390→        }
   391→
   392→        usedModel = candidate;
   393→        break; // success — stop failover
   394→      } catch (err) {
   395→        // ConfigurationError = provider not configured (skip silently).
   396→        // AIProviderError = upstream failure (try the next model).
   397→        // Anything else = unexpected; record and yield as an error.
   398→        if (err instanceof ConfigurationError) {
   399→          log.debug("skipping unconfigured provider", {
   400→            provider: candidate.provider,
   401→          });
   402→          continue;
   403→        }
   404→        if (err instanceof AIProviderError) {
   405→          log.warn("provider failed; trying next in failover chain", {
   406→            failedProvider: candidate.provider,
   407→            failedModel: candidate.id,
   408→            error: err.message,
   409→          });
   410→          // Record health for the failed provider.
   411→          await recordProviderOutcome(candidate.provider, {
   412→            success: false,
   413→            latencyMs: Date.now() - startedAt,
   414→            error: err.message,
   415→          });
   416→          streamError = err;
   417→          continue;
   418→        }
   419→        // Unexpected error — record and yield, no failover.
   420→        streamError = err instanceof Error ? err : new Error(String(err));
   421→        log.error("unexpected stream error", { error: String(err) });
   422→        break;
   423→      }
   424→    }
   425→
   426→    const latencyMs = Date.now() - startedAt;
   427→
   428→    // If we never got a successful model, record an error message and yield.
   429→    if (!usedModel) {
   430→      const errMsg = streamError
   431→        ? streamError.message
   432→        : "All AI providers failed. Please try again later.";
   433→      yield* this.recordAndYieldError(params, errMsg);
   434→      return;
   435→    }
   436→
   437→    // Successful completion — record usage, deduct credits, persist assistant message.
   438→    const costCents = modelManager.computeCostCents(
   439→      usedModel.provider,
   440→      usedModel.id,
   441→      inputTokens,
   442→      outputTokens,
   443→    );
   444→
   445→    // Persist the assistant message.
   446→    const messageService = await createMessageService();
   447→    let assistantMessageId: string | null = null;
   448→    try {
   449→      const assistantMessage = await messageService.create(
   450→        conversationId,
   451→        userId,
   452→        {
   453→          role: "assistant",
   454→          content: accumulated || "",
   455→          provider: usedModel.provider,
   456→          model: usedModel.id,
   457→          inputTokens,
   458→          outputTokens,
   459→          totalTokens,
   460→          costCents,
   461→          latencyMs,
   462→          finishReason: finishReason ?? "stop",
   463→          parentMessageId,
   464→        },
   465→      );
   466→      assistantMessageId = assistantMessage.id;
   467→    } catch (err) {
   468→      log.error("failed to persist assistant message", { error: String(err) });
   469→    }
   470→
   471→    // Record usage in the `ai_usage` table (admin client — no RLS insert policy).
   472→    await this.recordUsage({
   473→      userId,
   474→      conversationId,
   475→      messageId: assistantMessageId,
   476→      provider: usedModel.provider,
   477→      model: usedModel.id,
   478→      inputTokens,
   479→      outputTokens,
   480→      totalTokens,
   481→      costCents,
   482→      latencyMs,
   483→      status: "success",
   484→    });
   485→
   486→    // Deduct credits (best-effort — never block the response).
   487→    if (costCents > 0) {
   488→      try {
   489→        const reason: CreditReason = {
   490→          provider: usedModel.provider,
   491→          model: usedModel.id,
   492→          conversationId,
   493→          messageId: assistantMessageId ?? undefined,
   494→          feature: "chat",
   495→        };
   496→        await createCreditsService().deduct(userId, costCents, reason);
   497→      } catch (err) {
   498→        // Insufficient credits post-call (race) or DB failure — log and
   499→        // continue. The user still gets their response.
   500→        log.warn("failed to deduct credits after stream", {
   501→          costCents,
   502→          error: String(err),
   503→        });
   504→      }
   505→    }
   506→
   507→    // Record provider health (success).
   508→    await recordProviderOutcome(usedModel.provider, {
   509→      success: true,
   510→      latencyMs,
   511→    });
   512→
   513→    log.info("chat stream completed", {
   514→      provider: usedModel.provider,
   515→      model: usedModel.id,
   516→      inputTokens,
   517→      outputTokens,
   518→      totalTokens,
   519→      costCents,
   520→      latencyMs,
   521→    });
   522→  }
   523→
   524→  /**
   525→   * Persist an error assistant message + record usage + update provider
   526→   * health, then yield an error chunk to the client. Used when all failover
   527→   * candidates fail.
   528→   */
   529→  private async *recordAndYieldError(
   530→    params: {
   531→      conversationId: string;
   532→      userId: string;
   533→      provider: AIProvider;
   534→      model: string;
   535→      parentMessageId: string | null;
   536→    },
   537→    errorMessage: string,
   538→  ): AsyncIterable<ChatStreamChunk> {
   539→    const { conversationId, userId, provider, model, parentMessageId } = params;
   540→    const log = logger.child({ userId, conversationId, provider, model });
   541→
   542→    try {
   543→      const messageService = await createMessageService();
   544→      await messageService.create(conversationId, userId, {
   545→        role: "assistant",
   546→        content: "",
   547→        provider,
   548→        model,
   549→        finishReason: "error",
   550→        errorMessage,
   551→        parentMessageId,
   552→      });
   553→    } catch (err) {
   554→      log.error("failed to persist error assistant message", {
   555→        error: String(err),
   556→      });
   557→    }
   558→
   559→    await this.recordUsage({
   560→      userId,
   561→      conversationId,
   562→      provider,
   563→      model,
   564→      status: "error",
   565→      errorMessage,
   566→    });
   567→
   568→    await recordProviderOutcome(provider, {
   569→      success: false,
   570→      error: errorMessage,
   571→    });
   572→
   573→    yield {
   574→      delta: "",
   575→      finish_reason: "error",
   576→    };
   577→  }
   578→
   579→  /**
   580→   * Insert a row into `ai_usage` (per-request usage log). Uses the admin
   581→   * client because the table has no RLS insert policy — the chat API
   582→   * records usage server-side after the stream completes.
   583→   */
   584→  private async recordUsage(input: {
   585→    userId: string;
   586→    conversationId: string;
   587→    messageId?: string | null;
   588→    provider: AIProvider;
   589→    model: string;
   590→    inputTokens?: number;
   591→    outputTokens?: number;
   592→    totalTokens?: number;
   593→    costCents?: number;
   594→    latencyMs?: number;
   595→    status: "success" | "error" | "timeout" | "rate_limited";
   596→    errorMessage?: string;
   597→  }): Promise<void> {
   598→    const insert: TablesInsert<"ai_usage"> = {
   599→      user_id: input.userId,
   600→      conversation_id: input.conversationId,
   601→      message_id: input.messageId ?? null,
   602→      provider: input.provider,
   603→      model: input.model,
   604→      input_tokens: input.inputTokens ?? 0,
   605→      output_tokens: input.outputTokens ?? 0,
   606→      total_tokens: input.totalTokens ?? 0,
   607→      cost_cents: input.costCents ?? 0,
   608→      latency_ms: input.latencyMs ?? null,
   609→      feature: "chat",
   610→      status: input.status,
   611→      error_message: input.errorMessage ?? null,
   612→    };
   613→
   614→    try {
   615→      const { error } = await this.admin.from("ai_usage").insert(insert);
   616→      if (error) {
   617→        logger.warn("ai_usage insert failed", {
   618→          userId: input.userId,
   619→          provider: input.provider,
   620→          errorCode: error.code,
   621→          errorMessage: error.message,
   622→        });
   623→      }
   624→    } catch (err) {
   625→      // Best-effort — usage logging must never break the chat response.
   626→      logger.warn("ai_usage insert threw", {
   627→        userId: input.userId,
   628→        provider: input.provider,
   629→        error: String(err),
   630→      });
   631→    }
   632→  }
   633→
   634→  // -----------------------------------------------------------------------
   635→  // Provider/model resolution + token estimation
   636→  // -----------------------------------------------------------------------
   637→
   638→  /**
   639→   * Resolve the provider + model + system prompt for a streamResponse call.
   640→   * Priority: explicit request → conversation row → env default.
   641→   */
   642→  private resolveProviderModel(
   643→    input: StreamResponseInput,
   644→    conversation: Conversation,
   645→  ): { provider: AIProvider; model: string; systemPrompt?: string } {
   646→    const providerStr =
   647→      input.provider ?? conversation.provider ?? env.ai.defaultProvider;
   648→    const modelStr =
   649→      input.model ?? conversation.model ?? env.ai.defaultModel;
   650→    const systemPrompt = input.systemPrompt ?? conversation.system_prompt ?? undefined;
   651→
   652→    return this.resolveProviderModelFromStrings(providerStr, modelStr, systemPrompt);
   653→  }
   654→
   655→  /**
   656→   * Coerce raw provider/model strings into validated {@link AIProvider} +
   657→   * catalog model id. Falls back to the env-configured default provider +
   658→   * model manager's default model when the inputs are missing or invalid.
   659→   */
   660→  private resolveProviderModelFromStrings(
   661→    providerStr: string | undefined,
   662→    modelStr: string | undefined,
   663→    systemPrompt?: string,
   664→  ): { provider: AIProvider; model: string; systemPrompt?: string } {
   665→    const VALID_PROVIDERS: AIProvider[] = [
   666→      "openai",
   667→      "anthropic",
   668→      "google",
   669→      "openrouter",
   670→      "deepseek",
   671→      "qwen",
   672→      "grok",
   673→    ];
   674→    const fallbackProvider = aiRegistry.getDefaultId();
   675→    const resolvedProvider =
   676→      providerStr && VALID_PROVIDERS.includes(providerStr as AIProvider)
   677→        ? (providerStr as AIProvider)
   678→        : fallbackProvider;
   679→    const resolvedModel = modelStr ?? modelManager.getDefault().id;
   680→    return { provider: resolvedProvider, model: resolvedModel, systemPrompt };
   681→  }
   682→
   683→  /**
   684→   * Estimate the token count for a messages array. Uses the classic 4-chars-
   685→   * per-token heuristic — accurate enough for context-window validation
   686→   * (the AI provider will reject the call if we're wrong, and the model
   687→   * catalog's limit is conservative).
   688→   */
   689→  private estimateTokens(messages: ChatMessage[], systemPrompt?: string): number {
   690→    const charsPerToken = 4;
   691→    const totalChars = messages.reduce(
   692→      (sum, m) => sum + (m.content?.length ?? 0),
   693→      0,
   694→    ) + (systemPrompt?.length ?? 0);
   695→    return Math.ceil(totalChars / charsPerToken);
   696→  }
   697→
   698→  /**
   699→   * Convert DB message rows into the {@link ChatMessage} array shape the AI
   700→   * facade expects. Filters out error assistant messages (finish_reason='error')
   701→   * so they don't pollute the context.
   702→   */
   703→  private buildMessagesFromRows(
   704→    rows: Message[],
   705→    systemPrompt: string | null,
   706→  ): ChatMessage[] {
   707→    const messages: ChatMessage[] = [];
   708→    if (systemPrompt) {
   709→      messages.push({ role: "system", content: systemPrompt });
   710→    }
   711→    for (const row of rows) {
   712→      // Skip error assistant messages — they're not useful context.
   713→      if (row.role === "assistant" && row.finish_reason === "error") continue;
   714→      // Skip empty assistant messages (e.g. a failed stream).
   715→      if (row.role === "assistant" && !row.content) continue;
   716→      messages.push({
   717→        role: row.role,
   718→        content: String(row.content ?? ""),
   719→      });
   720→    }
   721→    return messages;
   722→  }
   723→}

   725→/**
   726→ * Build the canonical {@link ChatService}. The service holds an admin
   727→ * Supabase client for the `ai_usage` inserts (the table has no RLS insert
   728→ * policy, so writes must come from the service role).
   729→ */
   730→export function createChatService(): ChatService {
   731→  const admin = createSupabaseAdminClient();
   732→  return new ChatService(admin);
   733→}
