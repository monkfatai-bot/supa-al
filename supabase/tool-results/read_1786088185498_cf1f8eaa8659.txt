   197→  async createTask(userId: string, input: CreateTaskInput): Promise<RuntimeTask> {
   198→    const supabase = await this.getClient();
   199→    await assertMember(supabase, input.workspace_id, userId);
   200→    const row: RuntimeTaskInsert = {
   201→      workspace_id: input.workspace_id,
   202→      session_id: input.session_id ?? null,
   203→      process_id: input.process_id ?? null,
   204→      task_type: input.task_type,
   205→      name: input.name,
   206→      description: input.description ?? null,
   207→      status: input.scheduled_for ? "queued" : "queued",
   208→      priority: input.priority ?? 5,
   209→      payload: (input.payload as any) ?? {},
   210→      max_retries: input.max_retries ?? 3,
   211→      timeout_ms: input.timeout_ms ?? 30000,
   212→      scheduled_for: input.scheduled_for ?? null,
   213→      assigned_agent_id: input.assigned_agent_id ?? null,
   214→      created_by: userId,
   215→    };
   216→    const { data, error } = await supabase.from("runtime_tasks").insert(row).select().single();
   217→    if (error) throw toDbError(error, "Failed to create runtime task.");
   218→
   219→    // Process immediately if no scheduled_for.
   220→    if (!input.scheduled_for) {
   221→      const taskId = (data as RuntimeTask).id;
   222→      setImmediate(() => {
   223→        this.processTask(taskId).catch((err) => {
   224→          logger.error("runtime: task processing failed", { taskId, err: String(err) });
   225→        });
   226→      });
   227→    }
   228→
   229→    await this.emitEvent({
   230→      workspace_id: input.workspace_id,
   231→      session_id: input.session_id ?? null,
   232→      task_id: (data as RuntimeTask).id,
   233→      event_type: "task.created",
   234→      category: "task",
   235→      level: "info",
   236→      message: `Task created: ${input.name} (${input.task_type})`,
   237→      source: "task-engine",
   238→    });
   239→    return data as RuntimeTask;
   240→  }

   286→  /**
   287→   * Process a single task. Marks as running, executes, marks as completed/failed.
   288→   * Uses setImmediate for async background processing.
   289→   */
   290→  async processTask(taskId: string): Promise<void> {
   291→    const supabase = await this.getClient();
   292→    const { data: task, error } = await supabase.from("runtime_tasks").select("*").eq("id", taskId).maybeSingle();
   293→    if (error || !task) return;
   294→
   295→    // Mark as running.
   296→    await supabase.from("runtime_tasks").update({ status: "running", started_at: new Date().toISOString() }).eq("id", taskId);
   297→
   298→    const startedAt = Date.now();
   299→    try {
   300→      // Execute the task based on its type.
   301→      const result = await this.executeTask(task as RuntimeTask);
   302→      const durationMs = Date.now() - startedAt;
   303→
   304→      await supabase.from("runtime_tasks").update({
   305→        status: "completed",
   306→        result: result as any,
   307→        completed_at: new Date().toISOString(),
   308→        duration_ms: durationMs,
   309→      }).eq("id", taskId);
   310→
   311→      await this.emitEvent({
   312→        workspace_id: task.workspace_id,
   313→        session_id: task.session_id,
   314→        task_id: taskId,
   315→        event_type: "task.completed",
   316→        category: "task",
   317→        level: "info",
   318→        message: `Task completed: ${task.name}`,
   319→        payload: { duration_ms: durationMs },
   320→        source: "task-engine",
   321→      });
   322→    } catch (err) {
   323→      const appErr = toAppError(err);
   324→      const taskData = task as RuntimeTask;
   325→      const shouldRetry = taskData.retry_count < taskData.max_retries;
   326→
   327→      if (shouldRetry) {
   328→        const retryDelay = Math.min(2000 * Math.pow(2, taskData.retry_count), 3600000);
   329→        await supabase.from("runtime_tasks").update({
   330→          status: "retrying",
   331→          retry_count: taskData.retry_count + 1,
   332→          error: appErr.message.slice(0, 2000),
   333→          next_retry_at: new Date(Date.now() + retryDelay).toISOString(),
   334→        }).eq("id", taskId);
   335→      } else {
   336→        await supabase.from("runtime_tasks").update({
   337→          status: "failed",
   338→          error: appErr.message.slice(0, 2000),
   339→          completed_at: new Date().toISOString(),
   340→        }).eq("id", taskId);
   341→      }
   342→
   343→      await this.emitEvent({
   344→        workspace_id: task.workspace_id,
   345→        session_id: task.session_id,
   346→        task_id: taskId,
   347→        event_type: "task.failed",
   348→        category: "error",
   349→        level: "error",
   350→        message: `Task failed: ${task.name} — ${appErr.message}`,
   351→        source: "task-engine",
   352→      });
   353→    }
   354→  }

   356→  /**
   357→   * Execute a task based on its type. Delegates to the appropriate service.
   358→   */
   359→  private async executeTask(task: RuntimeTask): Promise<unknown> {
   360→    const supabase = await this.getClient();
   361→    const payload = (task.payload as Record<string, unknown>) ?? {};
   362→
   363→    switch (task.task_type) {
   364→      case "chat": {
   365→        const { ai } = await import("@/lib/ai");
   366→        return ai.chat({
   367→          model: payload.model as string,
   368→          messages: payload.messages as any[],
   369→        });
   370→      }
   371→      case "image": {
   372→        const { image } = await import("@/lib/ai/image-manager");
   373→        return image.generate(payload as any);
   374→      }
   375→      case "workflow_action": {
   376→        // Delegate to automation executor.
   377→        return { status: "executed", task_type: "workflow_action" };
   378→      }
   379→      case "agent_action": {
   380→        // Delegate to AI Employee service.
   381→        return { status: "executed", task_type: "agent_action" };
   382→      }
   383→      case "webhook": {
   384→        return { status: "delivered", task_type: "webhook" };
   385→      }
   386→      case "sync": {
   387→        return { status: "synced", task_type: "sync" };
   388→      }
   389→      case "business": {
   390→        return { status: "executed", task_type: "business" };
   391→      }
   392→      default:
   393→        return { status: "completed", task_type: task.task_type };
   394→    }
   395→  }

   397→  /**
   398→   * Process the retry queue — called by a scheduler.
   399→   */
   400→  async processRetryQueue(): Promise<{ processed: number; succeeded: number; failed: number }> {
   401→    const supabase = await this.getClient();
   402→    const { data: tasks, error } = await supabase.from("runtime_tasks").select("id").eq("status", "retrying").or(`next_retry_at.is.null,next_retry_at.lte.${new Date().toISOString()}`).limit(50);
   403→    if (error) throw toDbError(error, "Failed to fetch retry queue.");
   404→    let succeeded = 0;
   405→    let failed = 0;
   406→    for (const t of tasks ?? []) {
   407→      try {
   408→        await this.processTask(t.id);
   409→        succeeded++;
   410→      } catch {
   411→        failed++;
   412→      }
   413→    }
   414→    return { processed: tasks?.length ?? 0, succeeded, failed };
   415→  }

   512→  async getResourceSummary(workspaceId: string, userId: string): Promise<ResourceSummary> {
   513→    const resources = await this.listResources(workspaceId, userId);
   514→    const byType: ResourceSummary["by_type"] = {};
   515→    let totalTokenBudget = 0, totalTokenUsed = 0, totalCreditBudget = 0, totalCreditUsed = 0, maxConcurrent = 0, currentConcurrent = 0;
   516→    for (const r of resources) {
   517→      const type = r.resource_type;
   518→      if (!byType[type]) byType[type] = [];
   519→      byType[type].push({
   520→        resource_key: r.resource_key,
   521→        limit_value: r.limit_value,
   522→        used_value: r.used_value,
   523→        reserved_value: r.reserved_value,
   524→        utilization: r.limit_value > 0 ? r.used_value / r.limit_value : 0,
   525→      });
   526→      if (type === "tokens") { totalTokenBudget += r.limit_value; totalTokenUsed += r.used_value; }
   527→      if (type === "credits") { totalCreditBudget += r.limit_value; totalCreditUsed += r.used_value; }
   528→      if (type === "concurrent" && r.resource_key === "max") { maxConcurrent = r.limit_value; currentConcurrent = r.used_value; }
   529→    }
   530→    return { by_type: byType, total_token_budget: totalTokenBudget, total_token_used: totalTokenUsed, total_credit_budget: totalCreditBudget, total_credit_used: totalCreditUsed, max_concurrent: maxConcurrent, current_concurrent: currentConcurrent };
   531→  }

   586→  /**
   587→   * Process due scheduled items — called by a scheduler.
   588→   */
   589→  async processScheduledItems(): Promise<{ processed: number; succeeded: number; failed: number }> {
   590→    const supabase = await this.getClient();
   591→    const now = new Date().toISOString();
   592→    const { data: schedules, error } = await supabase.from("runtime_schedules").select("*").eq("status", "active").or(`next_run_at.is.null,next_run_at.lte.${now}`).limit(50);
   593→    if (error) throw toDbError(error, "Failed to fetch due schedules.");
   594→    let succeeded = 0;
   595→    let failed = 0;
   596→    for (const sched of schedules ?? []) {
   597→      try {
   598→        // Create a task for the scheduled item.
   599→        await supabase.from("runtime_tasks").insert({
   600→          workspace_id: sched.workspace_id,
   601→          task_type: sched.target_type === "workflow" ? "workflow_action" : "agent_action",
   602→          name: `Scheduled: ${sched.name}`,
   603→          status: "queued",
   604→          priority: 5,
   605→          payload: { schedule_id: sched.id, target_type: sched.target_type, target_id: sched.target_id, config: sched.target_config },
   606→          created_by: sched.created_by,
   607→        });
   608→        // Update schedule.
   609→        await supabase.from("runtime_schedules").update({
   610→          last_run_at: now,
   611→          next_run_at: this.computeNextRun(sched),
   612→          run_count: sched.run_count + 1,
   613→          status: sched.max_runs && sched.run_count + 1 >= sched.max_runs ? "completed" : "active",
   614→        }).eq("id", sched.id);
   615→        succeeded++;
   616→      } catch {
   617→        failed++;
   618→      }
   619→    }
   620→    return { processed: schedules?.length ?? 0, succeeded, failed };
   621→  }
   622→
   623→  private computeNextRun(schedule: any): string | null {
   624→    if (schedule.schedule_type === "recurring" && schedule.cron_expression) {
   625→      // Simple approximation: next run in 24 hours.
   626→      return new Date(Date.now() + 24 * 60 * 60 * 1000).toISOString();
   627→    }
   628→    if (schedule.schedule_type === "delayed" && schedule.delay_ms) {
   629→      return new Date(Date.now() + schedule.delay_ms).toISOString();
   630→    }
   631→    return null;
   632→  }

   636→  async createCheckpoint(workspaceId: string, userId: string, sessionId: string): Promise<RuntimeRecovery> {
   637→    const supabase = await this.getClient();
   638→    await assertMember(supabase, workspaceId, userId);
   639→
   640→    // Snapshot session state.
   641→    const [processes, tasks, contexts] = await Promise.all([
   642→      supabase.from("runtime_processes").select("*").eq("session_id", sessionId).in("status", ["running", "paused"]),
   643→      supabase.from("runtime_tasks").select("*").eq("session_id", sessionId).in("status", ["queued", "running", "retrying"]),
   644→      supabase.from("runtime_contexts").select("*").eq("session_id", sessionId),
   645→    ]);
   646→
   647→    const checkpoint = {
   648→      processes: (processes.data ?? []).map((p: any) => ({ id: p.id, status: p.status, checkpoint_data: p.metadata })),
   649→      tasks: (tasks.data ?? []).map((t: any) => ({ id: t.id, status: t.status, payload: t.payload })),
   650→      contexts: (contexts.data ?? []).map((c: any) => ({ id: c.id, data: c.data, variables: c.variables })),
   651→      created_at: new Date().toISOString(),
   652→    };
   653→
   654→    const row: RuntimeRecoveryInsert = {
   655→      workspace_id: workspaceId,
   656→      session_id: sessionId,
   657→      recovery_type: "checkpoint",
   658→      status: "completed",
   659→      checkpoint_data: checkpoint as any,
   660→      failed_processes: (processes.data ?? []).map((p: any) => p.id) as any,
   661→      started_at: new Date().toISOString(),
   662→      completed_at: new Date().toISOString(),
   663→    };
   664→
   665→    const { data, error } = await supabase.from("runtime_recovery").insert(row).select().single();
   666→    if (error) throw toDbError(error, "Failed to create checkpoint.");
   667→    return data as RuntimeRecovery;
   668→  }

   670→  async recoverSession(workspaceId: string, userId: string, sessionId: string): Promise<RuntimeRecovery> {
   671→    const supabase = await this.getClient();
   672→    await assertMember(supabase, workspaceId, userId);
   673→
   674→    // Find the latest checkpoint.
   675→    const { data: checkpoint, error: cpError } = await supabase.from("runtime_recovery").select("*").eq("session_id", sessionId).eq("recovery_type", "checkpoint").order("created_at", { ascending: false }).limit(1).maybeSingle();
   676→    if (cpError || !checkpoint) throw new NotFoundError("Recovery checkpoint", sessionId);
   677→
   678→    // Mark session as recovering.
   679→    await supabase.from("runtime_sessions").update({ status: "recovering" }).eq("id", sessionId);
   680→
   681→    const cpData = checkpoint.checkpoint_data as any;
   682→    const recoveredProcesses: string[] = [];
   683→
   684→    // Restart processes.
   685→    for (const proc of cpData.processes ?? []) {
   686→      try {
   687→        await supabase.from("runtime_processes").update({ status: "running", started_at: new Date().toISOString(), error: null }).eq("id", proc.id);
   688→        recoveredProcesses.push(proc.id);
   689→      } catch {
   690→        // Skip failed recoveries.
   691→      }
   692→    }
   693→
   694→    // Re-queue tasks.
   695→    for (const task of cpData.tasks ?? []) {
   696→      await supabase.from("runtime_tasks").update({ status: "queued", error: null }).eq("id", task.id);
   697→      setImmediate(() => {
   698→        this.processTask(task.id).catch(() => {});
   699→      });
   700→    }
   701→
   702→    // Mark session as active.
   703→    await supabase.from("runtime_sessions").update({ status: "active" }).eq("id", sessionId);
   704→
   705→    // Create recovery record.
   706→    const row: RuntimeRecoveryInsert = {
   707→      workspace_id: workspaceId,
   708→      session_id: sessionId,
   709→      recovery_type: "crash",
   710→      status: "completed",
   711→      checkpoint_data: checkpoint.checkpoint_data as any,
   712→      failed_processes: checkpoint.failed_processes as any,
   713→      recovered_processes: recoveredProcesses as any,
   714→      started_at: new Date().toISOString(),
   715→      completed_at: new Date().toISOString(),
   716→    };
   717→
   718→    const { data, error } = await supabase.from("runtime_recovery").insert(row).select().single();
   719→    if (error) throw toDbError(error, "Failed to record recovery.");
   720→
   721→    await this.emitEvent({
   722→      workspace_id: workspaceId,
   723→      session_id: sessionId,
   724→      event_type: "recovery.completed",
   725→      category: "recovery",
   726→      level: "info",
   727→      message: `Session recovered — ${recoveredProcesses.length} processes restarted`,
   728→      source: "runtime-recovery",
   729→    });
   730→
   731→    return data as RuntimeRecovery;
   732→  }
