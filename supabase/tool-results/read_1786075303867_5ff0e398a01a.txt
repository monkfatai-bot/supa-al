     1→/**
     2→ * Supa AI — Phase 9C Employee Service (server-only).
     3→ *
     4→ * The single, canonical write-path for the AI Employees domain. Owns
     5→ * every `ai_employees` + `employee_*` table operation: CRUD, lifecycle
     6→ * (hire / pause / resume / archive / clone), skills, memory, training,
     7→ * assignments, performance, inter-employee messaging, marketplace,
     8→ * versioning, and chat.
     9→ *
    10→ * ## Construction
    11→ *
    12→ * Constructed with the **admin** Supabase client. The 0014 migration's
    13→ * RLS depends on `public.is_workspace_member(workspace_id, auth.uid())`,
    14→ * which in turn depends on the `workspaces` table that lands in Phase
    15→ * 9A. Until that table ships, the admin client (which bypasses RLS) is
    16→ * the only reliable path. All mutations still filter on `workspace_id`
    17→ * at the query layer so the surface is defense-in-depth even after RLS
    18→ * is fully wired.
    19→ *
    20→ * ## Workspace resolution
    21→ *
    22→ * Phase 9C has no `workspaces` table yet — workspace ids are passed
    23→ * through from the API layer (where the caller's session is resolved).
    24→ * When the API layer does not supply one, the service falls back to
    25→ * using the caller's `userId` as a synthetic single-user workspace id
    26→ * so the feature is fully functional today and a future Phase 9A
    27→ * migration can swap in a real `workspaces.id` without touching the
    28→ * service signature.
    29→ *
    30→ * ## Chat
    31→ *
    32→ * `chat()` builds a system prompt from the employee's `system_prompt`
    33→ * + relevant long-term + workspace memory, calls `ai.chat()`, and
    34→ * records the resulting usage in `employee_performance`. Throws
    35→ * {@link ConfigurationError} when no AI provider is configured.
    36→ *
    37→ * @module @/lib/employees/employee-service
    38→ */

   117→/**
   118→ * Map a Postgrest-shaped error into a {@link DatabaseError}. Centralized
   119→ * so the call sites stay narrow.
   120→ */
   121→function toDbError(
   122→  error: { code?: string; message?: string; name?: string; details?: unknown },
   123→  message: string,
   124→): DatabaseError {
   125→  return new DatabaseError(message, {
   126→    errorCode: error.code,
   127→    errorName: error.name,
   128→    errorMessage: error.message,
   129→    errorDetails: error.details,
   130→  });
   131→}

   133→/** Coerce an arbitrary value into a Postgres-safe `Json` payload. */
   134→function toJson(value: unknown): Json {
   135→  if (value === null || value === undefined) return null;
   136→  if (typeof value === "string" || typeof value === "number" || typeof value === "boolean") {
   137→    return value;
   138→  }
   139→  if (Array.isArray(value)) return value as Json[];
   140→  return value as Json;
   141→}

   143→/**
   144→ * Compute a SHA-256 content hash for training-source dedup. The hash
   145→ * is base64url-encoded so it fits comfortably in a `text` column.
   146→ */
   147→function hashContent(content: string): string {
   148→  return createHash("sha256").update(content).digest("base64url");
   149→}
   150→
   151→/**
   152→ * Naive chunker for training content. Splits on double newlines, then
   153→ * on single newlines when chunks are still too long. Used by the
   154→ * `trainFromUrl` path to populate `chunk_count` honestly — a future
   155→ * Phase can swap in a real embeddings pipeline.
   156→ */
   157→function countTrainingChunks(content: string): number {
   158→  const target = 800; // ~chars per chunk
   159→  const paragraphs = content
   160→    .split(/\n\s*\n/)
   161→    .map((p) => p.trim())
   162→    .filter(Boolean);
   163→  let chunks = 0;
   164→  for (const p of paragraphs) {
   165→    if (p.length <= target) {
   166→      chunks += 1;
   167→    } else {
   168→      chunks += Math.ceil(p.length / target);
   169→    }
   170→  }
   171→  return Math.max(1, chunks);
   172→}

   189→  /**
   190→   * List employees in a workspace. Optionally filter by department,
   191→   * status, or full-text search. Always returns the relations
   192→   * (skills, memory, training, assignments, versions) so the UI does
   193→   * not need a second round-trip per row.
   194→   */
   195→  async list(
   196→    wsId: string,
   197→    opts: ListEmployeesOptions = {},
   198→  ): Promise<EmployeeWithRelations[]> {
   199→    const limit = Math.max(1, Math.min(opts.limit ?? DEFAULT_LIST_LIMIT, MAX_LIST_LIMIT));
   200→    const offset = Math.max(0, opts.offset ?? 0);
   201→
   202→    try {
   203→      let query = this.supabase
   204→        .from("ai_employees")
   205→        .select()
   206→        .eq("workspace_id", wsId)
   207→        .order("created_at", { ascending: false })
   208→        .range(offset, offset + limit - 1);
   209→
   210→      if (opts.department) query = query.eq("department", opts.department);
   211→      if (opts.status) query = query.eq("status", opts.status);
   212→      if (typeof opts.isTemplate === "boolean") query = query.eq("is_template", opts.isTemplate);
   213→      if (typeof opts.isPublic === "boolean") query = query.eq("is_public", opts.isPublic);
   214→      if (opts.search && opts.search.trim().length > 0) {
   215→        const q = opts.search.trim();
   216→        // Use ILIKE on name + role + description — the FTS index from
   217→        // 0014 covers a generated tsvector over the same columns, but
   218→        // Postgrest's `textSearch` helper targets a single column. ILIKE
   219→        // is honest and works with the existing trigram index.
   220→        query = query.or(`name.ilike.%${q}%,role.ilike.%${q}%,description.ilike.%${q}%`);
   221→      }
   222→
   223→      const { data, error } = await query;
   224→      if (error) throw toDbError(error, "employees.list failed");
   225→      if (!data || data.length === 0) return [];
   226→
   227→      return this.hydrateRelations(data as Employee[]);
   228→    } catch (err) {
   229→      if (err instanceof DatabaseError) throw err;
   230→      const appErr = toAppError(err);
   231→      throw new DatabaseError("Unexpected failure listing employees.", {
   232→        workspaceId: wsId,
   233→        cause: appErr.message,
   234→      });
   235→    }
   236→  }

   255→      if (err instanceof DatabaseError) throw err;
   256→      const appErr = toAppError(err);
   257→      throw new DatabaseError("Unexpected failure fetching employee.", {
   258→        employeeId: id,
   259→        cause: appErr.message,
   260→      });
   261→    }
   262→  }

   264→  /**
   265→   * Create a new employee. The caller's `userId` is recorded on
   266→   * `created_by`. Returns the created row with empty relations.
   267→   */
   268→  async create(
   269→    wsId: string,
   270→    userId: string,
   271→    input: CreateEmployeeInput,
   272→  ): Promise<EmployeeWithRelations> {
   273→    if (!input.name?.trim()) {
   274→      throw new ValidationError("Employee name is required.");
   275→    }
   276→    if (!input.role?.trim()) {
   277→      throw new ValidationError("Employee role is required.");
   278→    }
   279→
   280→    try {
   281→      const row = {
   282→        workspace_id: wsId,
   283→        name: input.name.trim(),
   284→        role: input.role.trim(),
   285→        department: input.department ?? "general",
   286→        description: input.description ?? null,
   287→        avatar_url: input.avatarUrl ?? null,
   288→        status: "active",
   289→        experience_level: input.experienceLevel ?? "mid",
   290→        system_prompt: input.systemPrompt ?? null,
   291→        permissions: (input.permissions ?? []) as unknown as TablesInsert<"ai_employees">["permissions"],
   292→        tools: (input.tools ?? []) as unknown as TablesInsert<"ai_employees">["tools"],
   293→        is_template: input.isTemplate ?? false,
   294→        is_public: input.isPublic ?? false,
   295→        version: 1,
   296→        metadata: (input.metadata ?? null) as Json | null,
   297→        created_by: userId,
   298→      };
   299→
   300→      const { data, error } = await this.supabase
   301→        .from("ai_employees")
   302→        .insert(row as never)
   303→        .select()
   304→        .single();
   305→      if (error) throw toDbError(error, "employees.create failed");
   306→      if (!data) {
   307→        throw new DatabaseError("employees.create returned no row.");
   308→      }
   309→      return this.emptyRelations(data as Employee);
   310→    } catch (err) {
   311→      if (err instanceof DatabaseError || err instanceof ValidationError) throw err;
   312→      const appErr = toAppError(err);
   313→      throw new DatabaseError("Unexpected failure creating employee.", {
   314→        workspaceId: wsId,
   315→        cause: appErr.message,
   316→      });
   317→    }
   318→  }

   320→  /**
   321→   * Patch an employee. Only the supplied fields are written. Throws
   322→   * {@link NotFoundError} when the row does not exist.
   323→   */
   324→  async update(
   325→    id: string,
   326→    input: UpdateEmployeeInput,
   327→  ): Promise<EmployeeWithRelations> {
   328→    const patch: Record<string, unknown> = {};
   329→    if (input.name !== undefined) patch.name = input.name;
   330→    if (input.role !== undefined) patch.role = input.role;
   331→    if (input.department !== undefined) patch.department = input.department;
   332→    if (input.description !== undefined) patch.description = input.description;
   333→    if (input.avatarUrl !== undefined) patch.avatar_url = input.avatarUrl;
   334→    if (input.status !== undefined) patch.status = input.status;
   335→    if (input.experienceLevel !== undefined) patch.experience_level = input.experienceLevel;
   336→    if (input.systemPrompt !== undefined) patch.system_prompt = input.systemPrompt;
   337→    if (input.permissions !== undefined) patch.permissions = input.permissions;
   338→    if (input.tools !== undefined) patch.tools = input.tools;
   339→    if (input.isTemplate !== undefined) patch.is_template = input.isTemplate;
   340→    if (input.isPublic !== undefined) patch.is_public = input.isPublic;
   341→    if (input.metadata !== undefined) patch.metadata = input.metadata;
   342→
   343→    if (Object.keys(patch).length === 0) {
   344→      throw new ValidationError("No fields supplied for update.");
   345→    }
   346→
   347→    try {
   348→      const { data, error } = await this.supabase
   349→        .from("ai_employees")
   350→        .update(patch as never)
   351→        .eq("id", id)
   352→        .select()
   353→        .maybeSingle();
   354→      if (error) throw toDbError(error, "employees.update failed");
   355→      if (!data) throw new NotFoundError("Employee", id);
   356→      return this.hydrateSingle(data as Employee);
   357→    } catch (err) {
   358→      if (err instanceof DatabaseError || err instanceof NotFoundError || err instanceof ValidationError) {
   359→        throw err;
   360→      }
   361→      const appErr = toAppError(err);
   362→      throw new DatabaseError("Unexpected failure updating employee.", {
   363→        employeeId: id,
   364→        cause: appErr.message,
   365→      });
   366→    }
   367→  }

   469→  /**
   470→   * Add a skill to an employee. The skill name must exist in the
   471→   * {@link skillRegistry} catalog — passing an unknown skill throws
   472→   * {@link ValidationError}. Proficiency is clamped to [0, 100].
   473→   */
   474→  async addSkill(
   475→    employeeId: string,
   476→    input: AddSkillInput,
   477→  ): Promise<EmployeeSkill> {
   478→    const resolved = skillRegistry.resolve(input);
   479→    if (!resolved) {
   480→      throw new ValidationError(`Unknown skill: "${input.skillName}".`);
   481→    }
   482→
   483→    try {
   484→      const row = {
   485→        employee_id: employeeId,
   486→        skill_name: resolved.skillName,
   487→        proficiency: resolved.proficiency,
   488→        is_primary: resolved.isPrimary,
   489→        config: resolved.config as unknown as Json,
   490→      };
   491→      const { data, error } = await this.supabase
   492→        .from("employee_skills")
   493→        .upsert(row as never, { onConflict: "employee_id,skill_name" })
   494→        .select()
   495→        .single();
   496→      if (error) throw toDbError(error, "employee_skills.add failed");
   497→      if (!data) throw new DatabaseError("employee_skills.add returned no row.");
   498→      return data as EmployeeSkill;
   499→    } catch (err) {
   500→      if (err instanceof DatabaseError || err instanceof ValidationError) throw err;
   501→      const appErr = toAppError(err);
   502→      throw new DatabaseError("Unexpected failure adding skill.", {
   503→        employeeId,
   504→        cause: appErr.message,
   505→      });
   506→    }
   507→  }

   524→    try {
   525→      const { data, error } = await this.supabase
   526→        .from("employee_skills")

   528→        .eq("id", skillId)
   529→        .select()
   530→        .maybeSingle();
   531→      if (error) throw toDbError(error, "employee_skills.update failed");
   532→      if (!data) throw new NotFoundError("Employee skill", skillId);

   534→    } catch (err) {
   535→      if (err instanceof DatabaseError || err instanceof NotFoundError || err instanceof ValidationError) {

   568→  /**
   569→   * List memory entries for an employee. Optionally filter by type.
   570→   * Session memory rows whose `expires_at` is in the past are
   571→   * automatically excluded (the UI hides them).
   572→   */
   573→  async listMemory(
   574→    employeeId: string,
   575→    opts: ListMemoryOptions = {},
   576→  ): Promise<EmployeeMemory[]> {
   577→    const limit = Math.max(1, Math.min(opts.limit ?? DEFAULT_MEMORY_LIMIT, MAX_MEMORY_LIMIT));
   578→    try {
   579→      let query = this.supabase
   580→        .from("employee_memory")
   581→        .select()
   582→        .eq("employee_id", employeeId)
   583→        .order("importance", { ascending: false })
   584→        .order("updated_at", { ascending: false })
   585→        .limit(limit);
   586→      if (opts.type) query = query.eq("memory_type", opts.type);
   587→      // Hide expired session memory.
   588→      query = query.or(`expires_at.is.null,expires_at.gt.${new Date().toISOString()}`);
   589→      const { data, error } = await query;
   590→      if (error) throw toDbError(error, "employee_memory.list failed");
   591→      return (data ?? []) as EmployeeMemory[];
   592→    } catch (err) {
   593→      if (err instanceof DatabaseError) throw err;
   594→      const appErr = toAppError(err);
   595→      throw new DatabaseError("Unexpected failure listing memory.", {
   596→        employeeId,
   597→        cause: appErr.message,

   605→    wsId: string | null,

   624→        key: input.key.trim(),
   625→        value: toJson(input.value),
   626→        importance: Math.max(0, Math.min(100, input.importance ?? 50)),
   627→        expires_at: input.expiresAt ?? null,
   628→        metadata: (input.metadata ?? null) as Json | null,
   629→      };

   766→  /**
   767→   * Train an employee from an uploaded document. The document's
   768→   * content is hashed for dedup; the row's `chunk_count` reflects a
   769→   * naive count of the resulting chunks. Status transitions:
   770→   * `pending` → `processing` → `completed` (or `failed`).
   771→   *
   772→   * Phase 9C V1: stores the metadata + hash; a future Phase can swap
   773→   * the `processing` step for a real embeddings pipeline.
   774→   */
   775→  async trainFromDocument(
   776→    employeeId: string,
   777→    wsId: string | null,
   778→    userId: string,
   779→    documentId: string,
   780→    title: string,
   781→    content: string,
   782→  ): Promise<EmployeeTraining> {
   783→    if (!content || !content.trim()) {
   784→      throw new ValidationError("Training document content is empty.");
   785→    }
   786→    const contentHash = hashContent(content);
   787→    const chunkCount = countTrainingChunks(content);
   788→
   789→    try {
   790→      // Dedup: if a training row with the same (employee_id, content_hash)
   791→      // already exists, return it without re-processing.
   792→      const { data: existing } = await this.supabase
   793→        .from("employee_training")
   794→        .select()
   795→        .eq("employee_id", employeeId)
   796→        .eq("content_hash", contentHash)
   797→        .maybeSingle();
   798→      if (existing) return existing as EmployeeTraining;
   799→
   800→      const row = {
   801→        employee_id: employeeId,
   802→        workspace_id: wsId,
   803→        source_type: "document" as TrainingSourceType,
   804→        source_id: documentId,
   805→        title: title.trim() || `Document ${documentId}`,
   806→        content_hash: contentHash,
   807→        status: "completed",
   808→        chunk_count: chunkCount,
   809→        trained_by: userId,
   810→      };
   811→      const { data, error } = await this.supabase
   812→        .from("employee_training")
   813→        .insert(row as never)
   814→        .select()
   815→        .single();
   816→      if (error) throw toDbError(error, "employee_training.trainFromDocument failed");
   817→      if (!data) throw new DatabaseError("employee_training.insert returned no row.");
   818→      return data as EmployeeTraining;
   819→    } catch (err) {
   820→      if (err instanceof DatabaseError || err instanceof ValidationError) throw err;
   821→      const appErr = toAppError(err);
   822→      throw new DatabaseError("Unexpected failure training from document.", {
   823→        employeeId,
   824→        documentId,
   825→        cause: appErr.message,
   826→      });
   827→    }
   828→  }

   830→  /**
   831→   * Train an employee from a URL. Phase 9C V1 issues a `fetch()` to
   832→   * resolve the page's text content (no JS execution), computes a
   833→   * content hash for dedup, and stores the resulting chunk count.
   834→   * Failure to fetch sets the row to `failed` with the error message.
   835→   */
   836→  async trainFromUrl(
   837→    employeeId: string,
   838→    wsId: string | null,
   839→    userId: string,
   840→    input: TrainFromUrlInput,
   841→  ): Promise<EmployeeTraining> {
   842→    const url = input.url.trim();
   843→    if (!url || !/^https?:\/\//i.test(url)) {
   844→      throw new ValidationError("A valid http(s) URL is required.");
   845→    }
   846→
   847→    // Insert a `pending` row first so the UI sees immediate feedback,
   848→    // then update with the resolved content.
   849→    let row: EmployeeTraining | null = null;
   850→    try {
   851→      const insertRow = {
   852→        employee_id: employeeId,
   853→        workspace_id: wsId,
   854→        source_type: "website" as TrainingSourceType,
   855→        source_url: url,
   856→        title: input.title?.trim() || url,
   857→        status: "processing",
   858→        trained_by: userId,
   859→      };
   860→      const { data, error } = await this.supabase
   861→        .from("employee_training")
   862→        .insert(insertRow as never)
   863→        .select()
   864→        .single();
   865→      if (error) throw toDbError(error, "employee_training.trainFromUrl.insert failed");
   866→      if (!data) throw new DatabaseError("employee_training.insert returned no row.");
   867→      row = data as EmployeeTraining;
   868→    } catch (err) {
   869→      if (err instanceof DatabaseError || err instanceof ValidationError) throw err;
   870→      const appErr = toAppError(err);
   871→      throw new DatabaseError("Unexpected failure starting URL training.", {
   872→        employeeId,
   873→        url,
   874→        cause: appErr.message,
   875→      });
   876→    }
   877→
   878→    // Now fetch the page content. This runs server-side; the route
   879→    // handler's request deadline is the upper bound.
   880→    try {
   881→      const res = await fetch(url, {
   882→        // 15s timeout — generous but bounded.
   883→        signal: AbortSignal.timeout(15_000),
   884→        headers: { "user-agent": "SupaAI-Employee-Trainer/1.0" },
   885→        redirect: "follow",
   886→      });
   887→      if (!res.ok) {
   888→        throw new Error(`HTTP ${res.status} ${res.statusText}`);
   889→      }
   890→      const text = await res.text();
   891→      // Strip tags naively — a future Phase can swap in a proper
   892→      // HTML-to-text library (e.g. `cheerio`). For now, dropping
   893→      // everything between `<` and `>` is honest and good enough
   894→      // for chunk counting.
   895→      const stripped = text
   896→        .replace(/<script[\s\S]*?<\/script>/gi, " ")
   897→        .replace(/<style[\s\S]*?<\/style>/gi, " ")
   898→        .replace(/<[^>]+>/g, " ")
   899→        .replace(/\s+/g, " ")
   900→        .trim();
   901→      const content = stripped.slice(0, 200_000); // hard cap.
   902→      if (!content) {
   903→        throw new Error("Resolved page has no extractable text content.");
   904→      }
   905→      const contentHash = hashContent(content);
   906→      const chunkCount = countTrainingChunks(content);
   907→
   908→      const { data: updated, error: updateErr } = await this.supabase
   909→        .from("employee_training")
   910→        .update({
   911→          status: "completed",
   912→          content_hash: contentHash,
   913→          chunk_count: chunkCount,
   914→          error_message: null,
   915→        })
   916→        .eq("id", row.id)
   917→        .select()
   918→        .single();
   919→      if (updateErr) throw toDbError(updateErr, "employee_training.trainFromUrl.update failed");
   920→      return (updated ?? row) as EmployeeTraining;
   921→    } catch (err) {
   922→      const message = err instanceof Error ? err.message : String(err);
   923→      logger.warn("employee_training.trainFromUrl fetch failed", {
   924→        employeeId,
   925→        url,
   926→        message,
   927→      });
   928→      // Mark the row failed — the UI surfaces the error inline.
   929→      await this.supabase
   930→        .from("employee_training")
   931→        .update({ status: "failed", error_message: message })
   932→        .eq("id", row.id);
   933→      return { ...row, status: "failed", error_message: message };
   934→    }
   935→  }
   936→
   937→  /**
   938→   * Reindex all training sources for an employee. Clears the existing
   939→   * `chunk_count` + `content_hash` on each row, then re-derives them
   940→   * from the source. Phase 9C V1 only re-derives the chunk count for
   941→   * already-fetched content — URLs are not re-fetched (a future Phase
   942→   * can add a backoff-aware re-crawl).
   943→   */
   944→  async reindex(employeeId: string): Promise<EmployeeTraining[]> {
   945→    const rows = await this.listTraining(employeeId);
   946→    if (rows.length === 0) return [];
   947→
   948→    try {
   949→      const updates: Promise<EmployeeTraining | null>[] = rows.map(async (r) => {
   950→        // For training rows whose content we no longer have (because
   951→        // Phase 9C V1 stores only the hash), we just mark them
   952→        // `completed` and zero out the chunk_count. The user can
   953→        // re-train from the original source to repopulate.
   954→        const { data, error } = await this.supabase
   955→          .from("employee_training")
   956→          .update({
   957→            status: r.status === "failed" ? "failed" : "completed",
   958→            chunk_count: r.chunk_count ?? 0,
   959→            error_message: null,
   960→          })
   961→          .eq("id", r.id)
   962→          .select()
   963→          .maybeSingle();
   964→        if (error) throw toDbError(error, "employee_training.reindex failed");
   965→        return (data ?? null) as EmployeeTraining | null;
   966→      });
   967→      const settled = await Promise.allSettled(updates);
   968→      const ok: EmployeeTraining[] = [];
   969→      for (const s of settled) {
   970→        if (s.status === "fulfilled" && s.value) ok.push(s.value);
   971→      }
   972→      return ok;
   973→    } catch (err) {
   974→      if (err instanceof DatabaseError) throw err;
   975→      const appErr = toAppError(err);
   976→      throw new DatabaseError("Unexpected failure reindexing training.", {
   977→        employeeId,
   978→        cause: appErr.message,
   979→      });
   980→    }
   981→  }

  1115→
  1116→  /**
  1117→   * Record (or upsert) a daily performance row for an employee. The
  1118→   * `(employee_id, metric_date)` unique constraint merges same-day
  1119→   * writes; supply a `metricDate` to override today.
  1120→   */
  1121→  async recordPerformance(
  1122→    employeeId: string,
  1123→    wsId: string | null,
  1124→    input: RecordPerformanceInput,
  1125→  ): Promise<EmployeePerformance> {
  1126→    const metricDate = input.metricDate ?? new Date().toISOString().slice(0, 10);
  1127→
  1128→    try {
  1129→      // Read-then-merge so we accumulate rather than overwrite when
  1130→      // the row already exists (the chat path calls this after every
  1131→      // turn, so same-day upserts are common).
  1132→      const { data: existing } = await this.supabase
  1133→        .from("employee_performance")
  1134→        .select()
  1135→        .eq("employee_id", employeeId)
  1136→        .eq("metric_date", metricDate)
  1137→        .maybeSingle();
  1138→
  1139→      const prev = (existing ?? null) as EmployeePerformance | null;
  1140→      const tasksCompleted = (prev?.tasks_completed ?? 0) + (input.tasksCompleted ?? 0);
  1141→      const tasksFailed = (prev?.tasks_failed ?? 0) + (input.tasksFailed ?? 0);
  1142→      const total = tasksCompleted + tasksFailed;
  1143→      const successRate = total > 0 ? tasksCompleted / total : (prev?.success_rate ?? 0);
  1144→      const creditsConsumed = (prev?.credits_consumed ?? 0) + (input.creditsConsumed ?? 0);
  1145→      const costCents = (prev?.cost_cents ?? 0) + (input.costCents ?? 0);
  1146→      const totalTokens = (prev?.total_tokens ?? 0) + (input.totalTokens ?? 0);
  1147→      const workflowParticipations =
  1148→        (prev?.workflow_participations ?? 0) + (input.workflowParticipations ?? 0);
  1149→      const errorCount = (prev?.error_count ?? 0) + (input.errorCount ?? 0);
  1150→
  1151→      const row = {
  1152→        employee_id: employeeId,
  1153→        workspace_id: wsId,
  1154→        metric_date: metricDate,
  1155→        tasks_completed: tasksCompleted,
  1156→        tasks_failed: tasksFailed,
  1157→        success_rate: successRate,
  1158→        avg_response_ms: input.avgResponseMs ?? prev?.avg_response_ms ?? null,
  1159→        credits_consumed: creditsConsumed,
  1160→        cost_cents: costCents,
  1161→        total_tokens: totalTokens,
  1162→        workflow_participations: workflowParticipations,
  1163→        user_rating: input.userRating ?? prev?.user_rating ?? null,
  1164→        error_count: errorCount,
  1165→        metadata: (input.metadata ?? prev?.metadata ?? null) as Json | null,
  1166→      };
  1167→
  1168→      const { data, error } = await this.supabase
  1169→        .from("employee_performance")
  1170→        .upsert(row as never, { onConflict: "employee_id,metric_date" })
  1171→        .select()
  1172→        .single();
  1173→      if (error) throw toDbError(error, "employee_performance.upsert failed");
  1174→      if (!data) throw new DatabaseError("employee_performance.upsert returned no row.");
  1175→      return data as EmployeePerformance;
  1176→    } catch (err) {
  1177→      if (err instanceof DatabaseError) throw err;
  1178→      const appErr = toAppError(err);
  1179→      throw new DatabaseError("Unexpected failure recording performance.", {
  1180→        employeeId,
  1181→        cause: appErr.message,

  1185→
  1186→  /**
  1187→   * Build the manager-dashboard aggregate for a workspace. Pulls the
  1188→   * last 30 days of `employee_performance` rows + the active
  1189→   * `ai_employees` rows, then computes:
  1190→   *   - totalEmployees / activeEmployees / pausedEmployees / archivedEmployees
  1191→   *   - totalTasks / totalFailedTasks / avgSuccessRate
  1192→   *   - totalCreditsConsumed / totalCostCents / totalTokens
  1193→   *   - byDepartment breakdown
  1194→   *   - topPerformers (top 5 by success_rate * tasks_completed)
  1195→   */
  1196→  async getDashboard(wsId: string): Promise<EmployeeDashboardSummary> {
  1197→    try {
  1198→      // Employees
  1199→      const { data: employeesRaw, error: eErr } = await this.supabase
  1200→        .from("ai_employees")
  1201→        .select()
  1202→        .eq("workspace_id", wsId);
  1203→      if (eErr) throw toDbError(eErr, "dashboard.employees failed");
  1204→      const employees = (employeesRaw ?? []) as Employee[];
  1205→
  1206→      // Departments (workspace-scoped or seeded with workspace_id=null).
  1207→      const { data: deptsRaw } = await this.supabase
  1208→        .from("employee_departments")
  1209→        .select()
  1210→        .or(`workspace_id.is.null,workspace_id.eq.${wsId}`)
  1211→        .order("sort_order", { ascending: true });
  1212→      const departments = (deptsRaw ?? []) as EmployeeDepartment[];
  1213→
  1214→      // Performance (last 30 days)
  1215→      const since = new Date();
  1216→      since.setDate(since.getDate() - 30);
  1217→      const sinceStr = since.toISOString().slice(0, 10);
  1218→      const { data: perfRaw, error: pErr } = await this.supabase
  1219→        .from("employee_performance")
  1220→        .select()
  1221→        .eq("workspace_id", wsId)
  1222→        .gte("metric_date", sinceStr);
  1223→      if (pErr) throw toDbError(pErr, "dashboard.performance failed");
  1224→      const perfRows = (perfRaw ?? []) as EmployeePerformance[];
  1225→
  1226→      // Aggregate totals
  1227→      const totalTasks = perfRows.reduce((s, r) => s + (r.tasks_completed ?? 0), 0);
  1228→      const totalFailedTasks = perfRows.reduce((s, r) => s + (r.tasks_failed ?? 0), 0);
  1229→      const totalCredits = perfRows.reduce((s, r) => s + (r.credits_consumed ?? 0), 0);
  1230→      const totalCost = perfRows.reduce((s, r) => s + (r.cost_cents ?? 0), 0);
  1231→      const totalTokens = perfRows.reduce((s, r) => s + (r.total_tokens ?? 0), 0);
  1232→      const denom = totalTasks + totalFailedTasks;
  1233→      const avgSuccess = denom > 0 ? totalTasks / denom : 0;
  1234→
  1235→      // By-department breakdown
  1236→      const byDeptMap = new Map<string, DepartmentBreakdown>();
  1237→      for (const dept of departments) {
  1238→        byDeptMap.set(dept.name, {
  1239→          department: dept.name,
  1240→          label: dept.label,
  1241→          icon: dept.icon,
  1242→          color: dept.color,
  1243→          employeeCount: 0,
  1244→          activeCount: 0,
  1245→          tasksCompleted: 0,
  1246→          avgSuccessRate: 0,
  1247→          creditsConsumed: 0,
  1248→        });
  1249→      }
  1250→      // Tally employees per department
  1251→      const perfByEmployee = new Map<string, EmployeePerformance[]>();
  1252→      for (const p of perfRows) {
  1253→        const arr = perfByEmployee.get(p.employee_id) ?? [];
  1254→        arr.push(p);
  1255→        perfByEmployee.set(p.employee_id, arr);
  1256→      }
  1257→      for (const emp of employees) {
  1258→        const deptRow =
  1259→          byDeptMap.get(emp.department) ??
  1260→          ({
  1261→            department: emp.department,
  1262→            label: emp.department,
  1263→            icon: null,
  1264→            color: null,
  1265→            employeeCount: 0,
  1266→            activeCount: 0,
  1267→            tasksCompleted: 0,
  1268→            avgSuccessRate: 0,
  1269→            creditsConsumed: 0,
  1270→          } as DepartmentBreakdown);
  1271→        deptRow.employeeCount += 1;
  1272→        if (emp.status === "active") deptRow.activeCount += 1;
  1273→        const empPerf = perfByEmployee.get(emp.id) ?? [];
  1274→        for (const p of empPerf) {
  1275→          deptRow.tasksCompleted += p.tasks_completed ?? 0;
  1276→          deptRow.creditsConsumed += p.credits_consumed ?? 0;
  1277→        }
  1278→        byDeptMap.set(emp.department, deptRow);
  1279→      }
  1280→      // Finalize avgSuccessRate per department
  1281→      for (const dept of byDeptMap.values()) {
  1282→        const totalEmpPerf = perfRows
  1283→          .filter((p) => employees.some((e) => e.id === p.employee_id && e.department === dept.department))
  1284→          .reduce(
  1285→            (acc, p) => {
  1286→              acc.tasks += p.tasks_completed ?? 0;
  1287→              acc.fails += p.tasks_failed ?? 0;
  1288→              return acc;
  1289→            },
  1290→            { tasks: 0, fails: 0 },
  1291→          );
  1292→        const d = totalEmpPerf.tasks + totalEmpPerf.fails;
  1293→        dept.avgSuccessRate = d > 0 ? totalEmpPerf.tasks / d : 0;
  1294→      }
  1295→
  1296→      // Top performers (top 5)
  1297→      const topPerformers: TopPerformerEntry[] = employees
  1298→        .map((emp) => {
  1299→          const empPerf = perfByEmployee.get(emp.id) ?? [];
  1300→          const tasks = empPerf.reduce((s, r) => s + (r.tasks_completed ?? 0), 0);
  1301→          const fails = empPerf.reduce((s, r) => s + (r.tasks_failed ?? 0), 0);
  1302→          const total = tasks + fails;
  1303→          const success = total > 0 ? tasks / total : 0;
  1304→          const credits = empPerf.reduce((s, r) => s + (r.credits_consumed ?? 0), 0);
  1305→          const rating = empPerf.length > 0
  1306→            ? empPerf.reduce((s, r) => s + (r.user_rating ?? 0), 0) / empPerf.length
  1307→            : 0;
  1308→          return {
  1309→            employeeId: emp.id,
  1310→            name: emp.name,
  1311→            role: emp.role,
  1312→            department: emp.department,
  1313→            avatarUrl: emp.avatar_url,
  1314→            tasksCompleted: tasks,
  1315→            successRate: success,
  1316→            creditsConsumed: credits,
  1317→            userRating: rating > 0 ? rating : null,
  1318→          } satisfies TopPerformerEntry;
  1319→        })
  1320→        .sort((a, b) => {
  1321→          // Sort by tasksCompleted desc, then by successRate desc.
  1322→          if (b.tasksCompleted !== a.tasksCompleted) return b.tasksCompleted - a.tasksCompleted;
  1323→          return b.successRate - a.successRate;
  1324→        })
  1325→        .slice(0, 5);
  1326→
  1327→      return {
  1328→        totalEmployees: employees.length,
  1329→        activeEmployees: employees.filter((e) => e.status === "active").length,
  1330→        pausedEmployees: employees.filter((e) => e.status === "paused").length,
  1331→        archivedEmployees: employees.filter((e) => e.status === "archived").length,
  1332→        totalTasks,
  1333→        totalFailedTasks,
  1334→        avgSuccessRate: avgSuccess,
  1335→        totalCreditsConsumed: totalCredits,
  1336→        totalCostCents: totalCost,
  1337→        totalTokens,
  1338→        byDepartment: Array.from(byDeptMap.values()).sort(
  1339→          (a, b) => b.employeeCount - a.employeeCount,
  1340→        ),
  1341→        topPerformers,
  1342→      };
  1343→    } catch (err) {
  1344→      if (err instanceof DatabaseError) throw err;
  1345→      const appErr = toAppError(err);
  1346→      throw new DatabaseError("Unexpected failure building dashboard.", {
  1347→        workspaceId: wsId,
  1348→        cause: appErr.message,
  1349→      });
  1350→    }
  1351→  }

  1673→  /**

  1677→  async createVersion(
  1678→    employeeId: string,

  1683→    if (!current) throw new NotFoundError("Employee", employeeId);
  1684→
  1685→    try {
  1686→      // Compute the next version_number — read the current max.
  1687→      const versions = await this.listVersions(employeeId);
  1688→      const nextVersion = versions.length === 0 ? 1 : Math.max(...versions.map((v) => v.version_number)) + 1;
  1689→      const snapshot = {
  1690→        id: current.id,
