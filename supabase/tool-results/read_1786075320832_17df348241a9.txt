   135→/** Normalized error shape consumed by the UI. */
   136→export interface EmployeeApiError {
   137→  message: string;
   138→  code?: string;
   139→  status?: number;
   140→}
   141→
   142→async function unwrapError(res: Response): Promise<EmployeeApiError> {
   143→  let raw: unknown = null;
   144→  try {
   145→    raw = await res.json();
   146→  } catch {
   147→    return {
   148→      message: `Request failed (${res.status}).`,
   149→      status: res.status,
   150→    };
   151→  }
   152→  const envelope = raw as ApiResponse<never>;
   153→  if (envelope && envelope.success === false && envelope.error) {
   154→    return {
   155→      message: envelope.error.message,
   156→      code: envelope.error.code,
   157→      status: res.status,
   158→    };
   159→  }
   160→  return { message: `Request failed (${res.status}).`, status: res.status };
   161→}
   162→
   163→/**
   164→ * Issue a JSON request and either return the typed `data` payload or
   165→ * throw a normalized {@link EmployeeApiError}.
   166→ */
   167→async function apiRequest<T>(
   168→  method: string,
   169→  url: string,
   170→  body?: unknown,
   171→  signal?: AbortSignal,
   172→): Promise<T> {
   173→  const init: RequestInit = {
   174→    method,
   175→    credentials: "include",
   176→    headers:
   177→      body !== undefined ? { "Content-Type": "application/json" } : undefined,
   178→    body: body !== undefined ? JSON.stringify(body) : undefined,
   179→    signal,
   180→  };
   181→  const res = await fetch(url, init);
   182→  if (!res.ok) {
   183→    throw await unwrapError(res);
   184→  }
   185→  const json = (await res.json()) as ApiResponse<T>;
   186→  if (!json.success) {
   187→    throw {
   188→      message: json.error?.message ?? "Unexpected response shape.",
   189→      code: json.error?.code,
   190→    } as EmployeeApiError;
   191→  }
   192→  return json.data;
   193→}

   199→/** GET `/api/employees` — paginated list of the caller's employees. */
   200→export function useEmployees(opts: ListEmployeesQuery = {}) {
   201→  return useQuery({
   202→    queryKey: employeeKeys.list(opts),
   203→    queryFn: () => {
   204→      const p = new URLSearchParams();
   205→      if (opts.search) p.set("search", opts.search);
   206→      if (opts.department) p.set("department", opts.department);
   207→      if (opts.status) p.set("status", opts.status);
   208→      if (opts.isTemplate !== undefined) p.set("isTemplate", String(opts.isTemplate));
   209→      if (opts.isPublic !== undefined) p.set("isPublic", String(opts.isPublic));
   210→      if (opts.limit) p.set("limit", String(opts.limit));
   211→      if (opts.offset) p.set("offset", String(opts.offset));
   212→      const qs = p.toString();
   213→      return apiRequest<{ employees: EmployeeWithRelations[] }>(
   214→        "GET",
   215→        `/api/employees${qs ? `?${qs}` : ""}`,
   216→      ).then((r) => r.employees);
   217→    },
   218→    placeholderData: (prev) => prev,
   219→    staleTime: 5 * 1000,
   220→  });
   221→}
