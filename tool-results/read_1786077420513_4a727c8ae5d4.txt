     2→-- SUPA AI — 0009_phase7_workspace.sql
     3→-- Phase 7 / 9: Workspace & Collaboration Platform.
     4→--
     5→-- Creates the `workspaces` table plus 11 workspace-scoped collaboration
     6→-- tables (members, folders, documents, document_versions, comments,
     7→-- knowledge_base, file_library, workspace_roles, workspace_activity,
     8→-- workspace_mentions, workspace_invitations).
     9→--
    10→-- CRITICAL: The `is_workspace_member(ws_id, user_id)` SQL function is created
    11→-- FIRST. Every RLS policy on every workspace-scoped table in the platform
    12→-- (including the Phase 9C `ai_employees` family from migration 0014) calls
    13→-- this function. Until Phase 9A shipped this function, those policies
    14→-- silently denied every row. Now they work for real.
    15→--
    16→-- Design principles (carried from 0001):
    17→--   • Idempotent — every statement is safe to re-run.
    18→--   • RLS on every new table. Default-deny. No `USING (true)` open policies.
    19→--   • Money is integer cents. Timestamps are timestamptz.
    20→--   • updated_at maintained by the shared set_updated_at() trigger from 0003.
    21→-- =============================================================================
    22→
    23→-- -----------------------------------------------------------------------------
    24→-- 0. is_workspace_member(ws_id, user_id) — SECURITY DEFINER
    25→--
    26→-- Returns true iff `user_id` is an `active` member of workspace `ws_id`.
    27→-- Marked SECURITY DEFINER so it can be inlined into RLS policies on tables
    28→-- the caller does not own (workspaces, workspace_members themselves).
    29→-- Search path pinned to `public` to prevent trojan-horse schema attacks.
    30→-- -----------------------------------------------------------------------------
    31→create or replace function public.is_workspace_member(ws_id uuid, user_id uuid)
    32→returns boolean
    33→language sql
    34→security definer
    35→set search_path = public
    36→as $$
    37→  select exists (
    38→    select 1
    39→    from public.workspace_members wm
    40→    where wm.workspace_id = ws_id
    41→      and wm.user_id = user_id
    42→      and wm.status = 'active'
    43→  );
    44→$$;

   115→-- Members of a workspace can read its member list. A user can always read their
   116→-- own rows so they can see pending invitations addressed to them.
   117→drop policy if exists "workspace_members_select_ws" on public.workspace_members;
   118→create policy "workspace_members_select_ws" on public.workspace_members for select
   119→  using (
   120→    user_id = auth.uid()
   121→    or public.is_workspace_member(workspace_id, auth.uid())
   122→  );
   123→
   124→-- A user can self-insert to accept an invitation; otherwise the workspace owner
   125→-- or an admin invites. We accept either path here — the service layer enforces
   126→-- the invite-token + role check before reaching this policy.
   127→drop policy if exists "workspace_members_insert_ws" on public.workspace_members;
   128→create policy "workspace_members_insert_ws" on public.workspace_members for insert
   129→  with check (
   130→    user_id = auth.uid()
   131→    or exists (
   132→      select 1 from public.workspaces w
   133→      where w.id = workspace_id
   134→        and (w.owner_id = auth.uid() or public.is_workspace_member(w.id, auth.uid()))
   135→    )
   136→  );
   137→
   138→drop policy if exists "workspace_members_update_ws" on public.workspace_members;
   139→create policy "workspace_members_update_ws" on public.workspace_members for update
   140→  using (
   141→    user_id = auth.uid()
   142→    or public.is_workspace_member(workspace_id, auth.uid())
   143→  );
   144→
   145→drop policy if exists "workspace_members_delete_ws" on public.workspace_members;

   148→    user_id = auth.uid()
